<script>

const canvas = document.getElementById("canvas");
const ctx = canvas.getContext("2d");

let W, H;

function resize() {
    W = canvas.width = window.innerWidth;
    H = canvas.height = window.innerHeight;
}

resize();
window.addEventListener("resize", resize);


/* =====================================================
   ESTRELLAS
===================================================== */

const estrellas = [];

for (let i = 0; i < 180; i++) {

    estrellas.push({
        x: Math.random(),
        y: Math.random() * 0.75,
        r: Math.random() * 1.5 + 0.3,
        brillo: Math.random() * Math.PI * 2
    });

}


/* =====================================================
   PARTÍCULAS DORADAS
===================================================== */

const particulas = [];

for (let i = 0; i < 80; i++) {

    particulas.push({
        x: Math.random(),
        y: Math.random(),
        velocidad: Math.random() * 0.002 + 0.0005,
        tamaño: Math.random() * 3 + 1,
        fase: Math.random() * Math.PI * 2
    });

}


/* =====================================================
   DIBUJAR ESTRELLAS
===================================================== */

function dibujarEstrellas(t) {

    estrellas.forEach(e => {

        const brillo =
            0.4 +
            0.6 * Math.sin(t * 0.001 + e.brillo);

        ctx.beginPath();

        ctx.arc(
            e.x * W,
            e.y * H,
            e.r,
            0,
            Math.PI * 2
        );

        ctx.fillStyle =
            `rgba(255,255,220,${brillo})`;

        ctx.fill();

    });

}


/* =====================================================
   PARTÍCULAS
===================================================== */

function dibujarParticulas(t) {

    particulas.forEach(p => {

        p.y -= p.velocidad;

        if (p.y < 0) {
            p.y = 1;
            p.x = Math.random();
        }

        const movimiento =
            Math.sin(t * 0.001 + p.fase) * 8;

        ctx.beginPath();

        ctx.arc(
            p.x * W + movimiento,
            p.y * H,
            p.tamaño,
            0,
            Math.PI * 2
        );

        ctx.fillStyle = "#ffd43b";

        ctx.shadowBlur = 15;
        ctx.shadowColor = "#ffcc33";

        ctx.fill();

        ctx.shadowBlur = 0;

    });

}


/* =====================================================
   TALLO
===================================================== */

function dibujarTallo(x, y, altura) {

    ctx.beginPath();

    ctx.moveTo(x, y);

    ctx.quadraticCurveTo(
        x - 10,
        y - altura / 2,
        x,
        y - altura
    );

    ctx.strokeStyle = "#4d8b31";
    ctx.lineWidth = 7;
    ctx.stroke();


    // Hojas

    ctx.fillStyle = "#5fa83a";

    ctx.beginPath();

    ctx.ellipse(
        x - 25,
        y - altura * 0.45,
        30,
        12,
        -0.5,
        0,
        Math.PI * 2
    );

    ctx.fill();


    ctx.beginPath();

    ctx.ellipse(
        x + 25,
        y - altura * 0.65,
        30,
        12,
        0.5,
        0,
        Math.PI * 2
    );

    ctx.fill();

}


/* =====================================================
   PÉTALO
===================================================== */

function petalo(cx, cy, angulo, radio, ancho) {

    ctx.save();

    ctx.translate(cx, cy);
    ctx.rotate(angulo);

    ctx.beginPath();

    ctx.ellipse(
        0,
        -radio,
        ancho,
        radio * 0.55,
        0,
        0,
        Math.PI * 2
    );

    ctx.fillStyle = "#ffd21f";

    ctx.shadowBlur = 15;
    ctx.shadowColor = "#ffb300";

    ctx.fill();

    ctx.restore();

}


/* =====================================================
   GIRASOL
===================================================== */

function dibujarGirasol(x, y, escala, rotacion) {

    ctx.save();

    ctx.translate(x, y);
    ctx.rotate(rotacion);

    const petalos = 20;

    /* Pétalos exteriores */

    for (let i = 0; i < petalos; i++) {

        const angulo =
            (Math.PI * 2 / petalos) * i;

        petalo(
            0,
            0,
            angulo,
            45 * escala,
            18 * escala
        );

    }


    /* Segunda capa */

    for (let i = 0; i < 12; i++) {

        const angulo =
            (Math.PI * 2 / 12) * i +
            0.15;

        petalo(
            0,
            0,
            angulo,
            35 * escala,
            15 * escala
        );

    }


    /* Centro */

    ctx.beginPath();

    ctx.arc(
        0,
        0,
        27 * escala,
        0,
        Math.PI * 2
    );

    ctx.fillStyle = "#5a3217";

    ctx.shadowBlur = 20;
    ctx.shadowColor = "#ffb300";

    ctx.fill();


    /* Semillas */

    ctx.shadowBlur = 0;

    for (let i = 0; i < 45; i++) {

        const angulo =
            i * 2.4;

        const radio =
            Math.sqrt(i / 45) * 23 * escala;

        const px =
            Math.cos(angulo) * radio;

        const py =
            Math.sin(angulo) * radio;

        ctx.beginPath();

        ctx.arc(
            px,
            py,
            1.5 * escala,
            0,
            Math.PI * 2
        );

        ctx.fillStyle = "#d98c18";

        ctx.fill();

    }

    ctx.restore();
}


/* =====================================================
   ÓRBITAS — toque de física
===================================================== */

function dibujarOrbitas(t) {

    const cx = W / 2;
    const cy = H * 0.67;

    ctx.save();

    ctx.translate(cx, cy);

    ctx.rotate(t * 0.00015);

    for (let i = 0; i < 3; i++) {

        ctx.beginPath();

        ctx.ellipse(
            0,
            0,
            180 + i * 60,
            45 + i * 15,
            i * 0.7,
            0,
            Math.PI * 2
        );

        ctx.strokeStyle =
            "rgba(255,210,40,0.22)";

        ctx.lineWidth = 1.5;

        ctx.stroke();

    }

    ctx.restore();

}


/* =====================================================
   ANIMACIÓN PRINCIPAL
===================================================== */

function animar(t) {

    /* Fondo */

    const gradiente =
        ctx.createLinearGradient(0, 0, 0, H);

    gradiente.addColorStop(0, "#020617");
    gradiente.addColorStop(0.55, "#07152b");
    gradiente.addColorStop(1, "#14230d");

    ctx.fillStyle = gradiente;

    ctx.fillRect(0, 0, W, H);


    dibujarEstrellas(t);

    dibujarParticulas(t);

    dibujarOrbitas(t);


    /* Posición de las flores */

    const centroX = W / 2;

    const suelo = H * 0.92;


    /* Tallos */

    dibujarTallo(
        centroX,
        suelo,
        H * 0.38
    );

    dibujarTallo(
        centroX - 170,
        suelo,
        H * 0.28
    );

    dibujarTallo(
        centroX + 170,
        suelo,
        H * 0.30
    );


    /* Flores */

    dibujarGirasol(
        centroX,
        suelo - H * 0.38,
        1.25,
        Math.sin(t * 0.001) * 0.03
    );

    dibujarGirasol(
        centroX - 170,
        suelo - H * 0.28,
        0.85,
        Math.sin(t * 0.0012) * 0.04
    );

    dibujarGirasol(
        centroX + 170,
        suelo - H * 0.30,
        0.9,
        Math.sin(t * 0.0011) * 0.04
    );


    requestAnimationFrame(animar);
}

requestAnimationFrame(animar);


/* =====================================================
   CORAZONES AL HACER CLICK
===================================================== */

document.addEventListener("click", e => {

    crearCorazon(e.clientX, e.clientY);

});


function crearCorazon(x, y) {

    const corazon = document.createElement("div");

    corazon.innerHTML = "💛";

    corazon.style.position = "fixed";
    corazon.style.left = x + "px";
    corazon.style.top = y + "px";
    corazon.style.fontSize = "25px";
    corazon.style.pointerEvents = "none";
    corazon.style.zIndex = "20";

    document.body.appendChild(corazon);


    corazon.animate(
        [
            {
                transform: "translateY(0) scale(1)",
                opacity: 1
            },

            {
                transform:
                    "translateY(-120px) scale(1.8)",
                opacity: 0
            }
        ],
        {
            duration: 1200,
            easing: "ease-out"
        }
    );


    setTimeout(() => {
        corazon.remove();
    }, 1200);

}

</script>
